#! /usr/bin/node

function logistic (time, midpoint, scale) {
return 1/(1+(Math.E**(-1*((time-midpoint)/scale))))
}




const range=7000000 // 7 million potential members in UK
const periods=52*3 // 3 years
const midpoint=periods/2
const scaling=periods/10
console.log("scaling: "+scaling)
for (i = 1; i<(periods+1); i+=1 ) {
let res=range*logistic(i,midpoint,scaling)
let resk=res/1000
console.log(i+","+res.toFixed(0)+","+resk.toFixed(0))
}
